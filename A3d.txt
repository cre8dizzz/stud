%{
#include <stdio.h>
#include <string.h>

#define MAX 200

struct symbol
{
    char lexeme[100];
    char type[30];
    int count;
};

struct symbol table[MAX];
int tableSize = 0;

void addToTable(char *lexeme, char *type)
{
    int i;
    for(i = 0; i < tableSize; i++)
    {
        if(strcmp(table[i].lexeme, lexeme) == 0 && strcmp(table[i].type, type) == 0)
        {
            table[i].count++;
            printf("%-12s --> %-22s (Already in Symbol Table)\n", lexeme, type);
            return;
        }
    }

    strcpy(table[tableSize].lexeme, lexeme);
    strcpy(table[tableSize].type, type);
    table[tableSize].count = 1;
    tableSize++;

    printf("%-12s --> %-22s (Added to Symbol Table)\n", lexeme, type);
}

void printTable()
{
    int i;
    printf("\n============================================================\n");
    printf("                        SYMBOL TABLE\n");
    printf("============================================================\n");
    printf("%-5s %-15s %-25s %-5s\n", "No.", "LEXEME", "TYPE", "COUNT");
    printf("------------------------------------------------------------\n");

    for(i = 0; i < tableSize; i++)
    {
        printf("%-5d %-15s %-25s %-5d\n",
               i + 1,
               table[i].lexeme,
               table[i].type,
               table[i].count);
    }

    printf("============================================================\n");
}
